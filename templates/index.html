// Application State const userId = "demo_user_" + Math.floor(Math.random() * 1000); let currentMode = "recovery_coach"; // DOM elements const chatMessages = document.getElementById('chatMessages'); const messageInput = document.getElementById('messageInput'); // Functions function addMessage(sender, text, isAI = true) { const div = document.createElement('div'); div.className = `chat-message mb-4 ${isAI ? '' : 'text-right'}`; div.innerHTML = `

${sender}

${text}

`; chatMessages.appendChild(div); chatMessages.scrollTop = chatMessages.scrollHeight; } async function sendMessage() { const text = messageInput.value.trim(); if (!text) return; addMessage('You', text, false); messageInput.value = ''; try { const response = await fetch('/api/chat', { method: 'POST', headers: {'Content-Type': 'application/json'}, body: JSON.stringify({message: text, user_id: userId, mode: currentMode}) }); const data = await response.json(); addMessage('Ashwas AI', data.response); } catch (e) { addMessage('Ashwas AI', 'Sorry, I am having trouble connecting. Please try again.'); } } async function triggerSupport() { const modal = document.getElementById('emergencyModal'); const content = document.getElementById('emergencyContent'); modal.classList.remove('hidden'); content.innerHTML = '

Generating emergency support...

'; try { const response = await fetch('/api/emergency-script', { method: 'POST', headers: {'Content-Type': 'application/json'}, body: JSON.stringify({user_id: userId, scenario: "panic", personal_context: {}}) }); const data = await response.json(); content.innerHTML = `

${data.response}

`; } catch (e) { content.innerHTML = '

Please contact emergency services (911 or 988) immediately.

'; } } function closeEmergency() { document.getElementById('emergencyModal').classList.add('hidden'); } function switchMode(mode) { currentMode = mode; document.querySelectorAll('.mode-btn').forEach(btn => { btn.classList.remove('active'); if(btn.dataset.mode === mode) btn.classList.add('active'); }); addMessage('Ashwas AI', `Switched to ${mode.replace('_', ' ')} mode.`); }
